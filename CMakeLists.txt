cmake_minimum_required(VERSION 3.20)
project(CCurate VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(CCURATE_GUI "Build Raven graphical interface" ON)
option(CCURATE_SANITIZE "Enable address/undefined sanitizers" OFF)
option(CCURATE_MEDIA "PDF, image, video and audio previews" ON)
if(WIN32)
 include(cmake/Windows.cmake)
 return()
endif()
find_package(OpenSSL REQUIRED)
find_package(Threads REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SQLITE REQUIRED IMPORTED_TARGET sqlite3)
pkg_check_modules(ARCHIVE REQUIRED IMPORTED_TARGET libarchive)
pkg_check_modules(MAGIC REQUIRED IMPORTED_TARGET libmagic)
if(CCURATE_MEDIA)
 pkg_check_modules(JPEG REQUIRED IMPORTED_TARGET libjpeg)
 add_executable(jpegfixture EXCLUDE_FROM_ALL tests/jpeg_fixture.c)
 target_link_libraries(jpegfixture PRIVATE PkgConfig::JPEG)
 pkg_check_modules(PREVIEW REQUIRED IMPORTED_TARGET gdk-pixbuf-2.0 poppler-glib libavformat libavcodec libavutil libswscale)
endif()
add_executable(ccurate src/main.c src/util.c src/activity.c src/catalog.c src/operations.c src/preview.c src/view.c src/platform/linux/source_access_linux.c)
target_include_directories(ccurate PRIVATE include)
target_compile_definitions(ccurate PRIVATE _GNU_SOURCE _FILE_OFFSET_BITS=64)
target_compile_options(ccurate PRIVATE -Wall -Wextra -Wformat=2 -Wshadow -Wmissing-prototypes -Wstrict-prototypes)
target_link_libraries(ccurate PRIVATE OpenSSL::Crypto PkgConfig::SQLITE PkgConfig::ARCHIVE PkgConfig::MAGIC Threads::Threads m)
if(CCURATE_MEDIA)
 target_sources(ccurate PRIVATE src/preview_jpeg.c)
 target_compile_definitions(ccurate PRIVATE CCURATE_MEDIA=1)
 target_link_libraries(ccurate PRIVATE PkgConfig::PREVIEW PkgConfig::JPEG)
endif()
if(CCURATE_GUI)
 set(RAVEN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
 set(RAVEN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
 set(RAVEN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
 set(RAVEN_BUILD_WEB OFF CACHE BOOL "" FORCE)
 set(RAVEN_FETCH_VULKAN_HEADERS OFF CACHE BOOL "" FORCE)
 add_subdirectory(vendor/Raven EXCLUDE_FROM_ALL)
 pkg_check_modules(FONTCONFIG REQUIRED IMPORTED_TARGET fontconfig)
 pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-2.0>=2.50)
 target_sources(ccurate PRIVATE
   src/desktop_open.c
   src/gui/gui.c
   src/gui/gui_startup.c
   src/gui/gui_controller.c
   src/gui/gui_browser.c
   src/gui/gui_navigation.c
   src/gui/gui_open.c
   src/gui/gui_jobs.c
   src/gui/gui_layout.c
   src/gui/gui_scroll.c
   src/gui/gui_render.c
   src/gui/gui_input.c
   src/gui/gui_theme.c
   src/gui/gui_activity.c
   src/gui/gui_choices.c
   src/gui/gui_fonts.c
   src/gui/gui_menu.c
   src/font_catalog.c
 )
 target_compile_definitions(ccurate PRIVATE CCURATE_GUI=1)
 target_link_libraries(ccurate PRIVATE raven::raven PkgConfig::FONTCONFIG PkgConfig::GIO)
endif()
if(CCURATE_SANITIZE)
 target_compile_options(ccurate PRIVATE -g -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(ccurate PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_executable(hashcheck tests/hash_driver.c src/util.c src/activity.c)
target_include_directories(hashcheck PRIVATE include)
target_compile_definitions(hashcheck PRIVATE _GNU_SOURCE _FILE_OFFSET_BITS=64)
target_link_options(hashcheck PRIVATE -Wl,--wrap=read)
target_link_libraries(hashcheck PRIVATE OpenSSL::Crypto PkgConfig::SQLITE Threads::Threads m)
add_test(NAME changing_hash COMMAND hashcheck)
add_executable(databasecheck tests/database_driver.c src/catalog.c src/util.c src/activity.c
 src/operations.c src/platform/linux/source_access_linux.c)
target_include_directories(databasecheck PRIVATE include)
target_compile_definitions(databasecheck PRIVATE _GNU_SOURCE _FILE_OFFSET_BITS=64)
target_link_libraries(databasecheck PRIVATE OpenSSL::Crypto PkgConfig::SQLITE Threads::Threads m)
if(CCURATE_SANITIZE)
 target_compile_options(databasecheck PRIVATE -g -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(databasecheck PRIVATE -fsanitize=address,undefined)
endif()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
 add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:ccurate>)
 add_test(NAME database_recovery COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/database_recovery.py $<TARGET_FILE:ccurate> $<TARGET_FILE:databasecheck>)
endif()
add_executable(previewcheck EXCLUDE_FROM_ALL tests/preview_driver.c src/preview.c src/util.c src/activity.c)
target_include_directories(previewcheck PRIVATE include)
target_compile_definitions(previewcheck PRIVATE _GNU_SOURCE _FILE_OFFSET_BITS=64)
target_link_libraries(previewcheck PRIVATE OpenSSL::Crypto PkgConfig::SQLITE PkgConfig::ARCHIVE PkgConfig::MAGIC Threads::Threads m)

 target_link_libraries(previewcheck PRIVATE PkgConfig::PREVIEW PkgConfig::JPEG)

install(TARGETS ccurate RUNTIME DESTINATION bin)
add_library(preview_policy_probe SHARED EXCLUDE_FROM_ALL tests/preview_limits_probe.c)
target_link_libraries(preview_policy_probe PRIVATE ${CMAKE_DL_LIBS})

target_include_directories(workercheck PRIVATE include)
target_compile_definitions(workercheck PRIVATE _GNU_SOURCE _FILE_OFFSET_BITS=64)
target_link_libraries(workercheck PRIVATE OpenSSL::Crypto PkgConfig::SQLITE Threads::Threads m)

 target_compile_options(workercheck PRIVATE -g -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(workercheck PRIVATE -fsanitize=address,undefined)
endif()

target_include_directories(activitycheck PRIVATE include)
target_compile_definitions(activitycheck PRIVATE _GNU_SOURCE _FILE_OFFSET_BITS=64)
target_link_libraries(activitycheck PRIVATE OpenSSL::Crypto PkgConfig::SQLITE Threads::Threads m)

 target_compile_options(activitycheck PRIVATE -g -fsanitize=address,undefined -fno-omit-frame-pointer)

target_include_directories(sortcheck PRIVATE include)
